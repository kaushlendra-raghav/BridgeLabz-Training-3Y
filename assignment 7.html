CSS Code

/* Attribute selectors: match input type and href beginning with https */
.selector-demo input[type="text"] { padding:10px; border:2px solid #9aa8b8; border-radius:6px; }
.selector-demo a[href^="https"] { color:#087443; font-weight:bold; }
/* Pseudo-classes: hover, focus and nth-child */
.selector-demo a:hover { color:#d94841; }
.selector-demo input:focus { outline:3px solid #8ecae6; }
.selector-list li:nth-child(2) { background:#eef6ff; }
/* Pseudo-elements: add decorative content before and after */
.selector-demo h2::before { content:"★ "; color:#e09f3e; }
.selector-demo h2::after { content:""; display:block; width:70px; height:3px; background:#2475a8; }
/* Combinators: descendant, child (>), adjacent sibling (+), general sibling (~) */
.selector-parent p { padding:5px; }
.selector-parent > p { border-left:3px solid #2475a8; }
.selector-note + ul { margin-top:8px; }
.sibling-target + p { color:#a33; }
.sibling-target ~ p { text-decoration:underline; }
